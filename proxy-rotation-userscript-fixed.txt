  function parseProxy(input) {
    if (typeof input === "object" && input !== null) {
      const scheme = String(input.scheme || "http").toLowerCase();
      return {
        id: `${scheme}://${input.host}:${input.port}`,
        label: `${input.host}:${input.port}`,
        host: String(input.host || "").trim(),
        port: Number(input.port),
        scheme,
        username: String(input.username || ""),
        password: String(input.password || "")
      };
    }

    const raw = String(input || "").trim();
    if (!raw) throw new Error("Пустая строка прокси");

    let value = raw;
    let scheme = "http";

    const schemeMatch = value.match(/^(https?|socks4|socks5):\/\//i);
    if (schemeMatch) {
      scheme = schemeMatch[1].toLowerCase();
      value = value.slice(schemeMatch[0].length);
    }

    const match = value.match(/^([^:\/]+):(\d+)(?::([^:]*):(.*))?$/);
    if (!match) {
      throw new Error(`Неверный формат прокси: ${raw}`);
    }

    const [, host, portRaw, username = "", password = ""] = match;
    const port = Number(portRaw);

    if (!host || !Number.isInteger(port) || port < 1 || port > 65535) {
      throw new Error(`Неверный host/port: ${raw}`);
    }

    return {
      id: `${scheme}://${host}:${port}`,
      label: `${host}:${port}`,
      host,
      port,
      scheme,
      username,
      password
    };
  }

  class ProxyExtensionBridge {
    constructor() {
      this.pending = new Map();
      this.counter = 0;
      this.connected = false;
      this.eventHandlers = new Set();

      this.onMessage = this.onMessage.bind(this);
      window.addEventListener("message", this.onMessage);
    }

    onMessage(event) {
      const data = event.data;
      if (!data || data.channel !== CONFIG.channel) return;
      if (data.direction !== "extension-to-userscript") return;

      if (data.type === "response") {
        const pending = this.pending.get(data.requestId);
        if (!pending) return;

        this.pending.delete(data.requestId);
        clearTimeout(pending.timer);

        if (data.ok === false) {
          pending.reject(new Error(data.error || data.result?.error || "Extension error"));
        } else {
          pending.resolve(data.result);
        }
        return;
      }

      if (data.type === "event") {
        for (const handler of this.eventHandlers) {
          try {
            handler(data.event, data.payload);
          } catch (error) {
            console.error("[Proxy Rotation] Bridge event handler error:", error);
          }
        }
      }
    }

    request(type, payload = null, timeoutMs = 15000) {
      const requestId = `pr-${Date.now()}-${++this.counter}`;

      return new Promise((resolve, reject) => {
        const timer = setTimeout(() => {
          this.pending.delete(requestId);
          reject(new Error(`Extension timeout: ${type}`));
        }, timeoutMs);

        this.pending.set(requestId, { resolve, reject, timer });

        window.postMessage({
          channel: CONFIG.channel,
          direction: "userscript-to-extension",
          type,
          requestId,
          payload
        }, "*");
      });
    }

    async connect() {
      const result = await this.request("hello", null, 5000);
      this.connected = true;
      return result;
    }

    async setProxy(proxy) {
      if (!this.connected) await this.connect();
      return await this.request("setProxy", { proxy }, 15000);
    }

    async getCurrentProxy() {
      if (!this.connected) await this.connect();
      return await this.request("getCurrentProxy", null, 5000);
    }

    async testProxy(timeoutMs = CONFIG.testTimeoutMs) {
      if (!this.connected) await this.connect();
      return await this.request("testProxy", { timeoutMs }, timeoutMs + 5000);
    }

    async disconnect() {
      this.connected = false;
    }

    onEvent(handler) {
      this.eventHandlers.add(handler);
      return () => this.eventHandlers.delete(handler);
    }

    destroy() {
      for (const pending of this.pending.values()) {
        clearTimeout(pending.timer);
        pending.reject(new Error("Bridge destroyed"));
      }

      this.pending.clear();
      this.eventHandlers.clear();
      window.removeEventListener("message", this.onMessage);
      this.connected = false;
    }
  }

  class ProxyRotationManager {
    constructor(proxyList, bridge) {
      this.bridge = bridge;
      this.proxyList = [];
      this.currentProxyIndex = 0;
      this.lastSwitchAt = 0;
      this.rotationIntervalMs = CONFIG.intervalMs;
      this.autoRotation = CONFIG.autoStart;
      this.timerId = null;
      this.destroyed = false;
      this.switchInProgress = false;
      this.errorCooldownUntil = 0;
      this.listeners = new Set();

      this.setProxyList(proxyList);

      this.unsubscribeBridge = this.bridge.onEvent((event, payload) => {
        if (event === "PROXY_ERROR") {
          this.handleError(payload);
        }
      });
    }

    log(...args) {
      console.log("[Proxy Rotation]", ...args);
    }

    error(...args) {
      console.error("[Proxy Rotation]", ...args);
    }

    onChange(handler) {
      this.listeners.add(handler);
      return () => this.listeners.delete(handler);
    }

    emit() {
      const state = this.getState();
      for (const handler of this.listeners) {
        try {
          handler(state);
        } catch (error) {
          this.error("UI listener error:", error);
        }
      }
    }

    setProxyList(list) {
      this.proxyList = [];

      for (const item of list) {
        try {
          this.proxyList.push(parseProxy(item));
        } catch (error) {
          this.error("Invalid proxy skipped:", error);
        }
      }

      if (this.proxyList.length === 0) {
        this.currentProxyIndex = 0;
      } else {
        this.currentProxyIndex = Math.min(
          Math.max(0, this.currentProxyIndex),
          this.proxyList.length - 1
        );
      }
    }

    addProxy(input) {
      const proxy = parseProxy(input);

      if (this.proxyList.some(item =>
        item.host === proxy.host &&
        Number(item.port) === Number(proxy.port) &&
        item.scheme === proxy.scheme &&
        item.username === proxy.username
      )) {
        throw new Error("Такой прокси уже есть в списке");
      }

      this.proxyList.push(proxy);
      this.saveState();
      this.emit();
      return proxy;
    }

    async deleteCurrentProxy() {
      if (this.proxyList.length === 0) return false;

      const wasCurrent = this.currentProxyIndex;

      if (this.proxyList.length === 1) {
        try {
          await this.bridge.request("clearProxy");
        } catch (error) {
          this.error("Failed to clear proxy before deleting the last entry:", error);
          return false;
        }

        this.proxyList.splice(0, 1);
        this.currentProxyIndex = 0;
        this.autoRotation = false;
        this.saveState();
        this.emit();
        return true;
      }

      const targetIndex = (wasCurrent + 1) % this.proxyList.length;
      const target = this.proxyList[targetIndex];

      try {
        await this.bridge.setProxy(target);
      } catch (error) {
        this.error("Failed to activate replacement proxy:", error);
        return false;
      }

      this.proxyList.splice(wasCurrent, 1);

      if (targetIndex > wasCurrent) {
        this.currentProxyIndex = targetIndex - 1;
      } else {
        this.currentProxyIndex = 0;
      }

      this.lastSwitchAt = Date.now();
      this.saveState();
      this.emit();
      return true;
    }

    get currentProxy() {
      return this.proxyList[this.currentProxyIndex] || null;
    }

    getState() {
      const elapsed = this.lastSwitchAt
        ? Math.max(0, Date.now() - this.lastSwitchAt)
        : 0;

      const remaining = this.autoRotation
        ? Math.max(0, this.rotationIntervalMs - elapsed)
        : 0;

      return {
        proxyCount: this.proxyList.length,
        currentProxyIndex: this.currentProxyIndex,
        currentProxy: this.currentProxy,
        lastSwitchAt: this.lastSwitchAt,
        rotationIntervalMs: this.rotationIntervalMs,
        autoRotation: this.autoRotation,
        remainingMs: remaining,
        switchInProgress: this.switchInProgress,
        connected: this.bridge.connected
      };
    }

    async loadState() {
      try {
        const raw = localStorage.getItem(CONFIG.storageKey);
        if (!raw) return;

        const state = JSON.parse(raw);

        if (Array.isArray(state.proxyList)) {
          this.setProxyList(state.proxyList);
        }

        if (Number.isInteger(state.currentProxyIndex)) {
          this.currentProxyIndex = state.currentProxyIndex;
        }

        if (Number.isFinite(state.lastSwitchAt)) {
          this.lastSwitchAt = state.lastSwitchAt;
        }

        if (Number.isFinite(state.rotationIntervalMs) && state.rotationIntervalMs >= 1000) {
          this.rotationIntervalMs = state.rotationIntervalMs;
        }

        if (typeof state.autoRotation === "boolean") {
          this.autoRotation = state.autoRotation;
        }

        if (this.proxyList.length > 0) {
          this.currentProxyIndex = Math.min(
            Math.max(0, this.currentProxyIndex),
            this.proxyList.length - 1
          );
        }
      } catch (error) {
        this.error("Failed to load state:", error);
      }
    }

    saveState() {
      try {
        localStorage.setItem(CONFIG.storageKey, JSON.stringify({
          proxyList: this.proxyList,
          currentProxyIndex: this.currentProxyIndex,
          lastSwitchAt: this.lastSwitchAt,
          rotationIntervalMs: this.rotationIntervalMs,
          autoRotation: this.autoRotation
        }));
      } catch (error) {
        this.error("Failed to save state:", error);
      }
    }

    async init() {
      if (this.destroyed) return;

      await this.loadState();

      this.log("Initialized");
      this.log("Current proxy:", this.currentProxyIndex + 1);
      this.log("Proxy count:", this.proxyList.length);
      this.log("Rotation interval:", this.rotationIntervalMs / 1000, "s");

      try {
        await this.bridge.connect();
        this.log("Companion extension connected");
      } catch (error) {
        this.error("Companion extension is unavailable:", error);
      }

      if (this.proxyList.length === 0) {
        this.error("Proxy list is empty");
        this.emit();
        return;
      }

      let currentMatchesExtension = false;

      try {
        const extensionState = await this.bridge.getCurrentProxy();
        const extProxy = extensionState?.proxy;

        if (extProxy && this.currentProxy) {
          currentMatchesExtension =
            extProxy.host === this.currentProxy.host &&
            Number(extProxy.port) === Number(this.currentProxy.port) &&
            extProxy.scheme === this.currentProxy.scheme;
        }
      } catch (error) {
        this.error("Could not read extension proxy state:", error);
      }

      if (!currentMatchesExtension) {
        this.log("Current proxy differs from extension state");
        await this.setProxy(this.currentProxyIndex, "INITIALIZATION");
      } else {
        this.log("Current proxy already active in extension");

        if (!this.lastSwitchAt) {
          this.lastSwitchAt = Date.now();
          this.saveState();
        }

        this.emit();
      }

      if (this.autoRotation) {
        const elapsed = Date.now() - this.lastSwitchAt;

        if (elapsed >= this.rotationIntervalMs) {
          await this.rotate("TIMER");
        }

        this.start();
      }
    }

    start() {
      this.stopTimerOnly();

      if (!this.autoRotation || this.proxyList.length < 2) {
        this.emit();
        return;
      }

      this.timerId = setInterval(() => {
        this.updateTimer();
      }, 1000);

      this.log("Next rotation in:", Math.ceil(this.getState().remainingMs / 1000) + "s");
      this.emit();
    }

    stop() {
      this.autoRotation = false;
      this.saveState();
      this.stopTimerOnly();
      this.emit();
      this.log("Automatic rotation stopped");
    }

    startAuto() {
      this.autoRotation = true;

      if (!this.lastSwitchAt) {
        this.lastSwitchAt = Date.now();
      }

      this.saveState();
      this.start();
      this.log("Automatic rotation started");
    }

    stopTimerOnly() {
      if (this.timerId !== null) {
        clearInterval(this.timerId);
        this.timerId = null;
      }
    }

    updateTimer() {
      if (this.destroyed || !this.autoRotation) return;

      const remaining = Math.max(
        0,
        this.rotationIntervalMs - (Date.now() - this.lastSwitchAt)
      );

      if (remaining <= 0 && !this.switchInProgress) {
        this.rotate("TIMER").catch(error => {
          this.error("Timer rotation failed:", error);
        });
      }

      this.emit();
    }

    async rotate(reason) {
      if (this.proxyList.length < 2) {
        this.error("Rotation requires at least 2 proxies");
        return false;
      }

      const nextIndex =
        (this.currentProxyIndex + 1) % this.proxyList.length;

      return await this.setProxy(nextIndex, reason);
    }

    async next() {
      return await this.rotate("MANUAL_NEXT");
    }

    async previous() {
      if (this.proxyList.length < 2) return false;

      const previousIndex =
        (this.currentProxyIndex - 1 + this.proxyList.length) %
        this.proxyList.length;

      return await this.setProxy(previousIndex, "MANUAL_PREVIOUS");
    }

    async setProxy(index, reason = "MANUAL_NEXT") {
      if (this.destroyed) return false;

      if (!Number.isInteger(index) || index < 0 || index >= this.proxyList.length) {
        this.error("Invalid proxy index:", index);
        return false;
      }

      if (this.proxyList.length > 1 && index === this.currentProxyIndex && reason !== "INITIALIZATION") {
        this.log("Requested proxy is already active; skipped duplicate selection");
        return false;
      }

      if (this.switchInProgress) {
        this.log("Switch already in progress; request ignored");
        return false;
      }

      const target = this.proxyList[index];
      const oldIndex = this.currentProxyIndex;

      this.switchInProgress = true;
      this.emit();

      this.log("Switching proxy...");
      this.log(`Reason: ${reason}`);

      try {
        await this.bridge.setProxy(target);

        this.currentProxyIndex = index;
        this.lastSwitchAt = Date.now();
        this.saveState();

        this.log(`Proxy changed: ${oldIndex + 1} → ${index + 1}`);
        this.log(`Reason: ${reason}`);
        this.log("Proxy activated successfully");

        this.emit();
        return true;
      } catch (error) {
        this.error("ERROR");
        this.error("Failed to activate proxy");
        this.error("Reason:", error);

        this.emit();
        return false;
      } finally {
        this.switchInProgress = false;
        this.emit();
      }
    }

    async testProxy() {
      try {
        const result = await this.bridge.testProxy(CONFIG.testTimeoutMs);
        this.emit();
        return result;
      } catch (error) {
        this.error("Proxy test error:", error);
        return {
          ok: false,
          status: "ERROR",
          error: error.message
        };
      }
    }

    async handleError(payload) {
      this.error("Extension reported proxy error:", payload);
      this.emit();
    }

    setIntervalMs(seconds) {
      const value = Number(seconds);

      if (!Number.isFinite(value) || value < 1) {
        throw new Error("Интервал должен быть не меньше 1 секунды");
      }

      this.rotationIntervalMs = Math.floor(value * 1000);
      this.lastSwitchAt = Date.now();
      this.saveState();

      if (this.autoRotation) {
        this.start();
      }

      this.emit();
    }

    destroy() {
      if (this.destroyed) return;

      this.destroyed = true;
      this.stopTimerOnly();
      this.unsubscribeBridge?.();
      this.bridge.disconnect();
      this.listeners.clear();

      this.log("Destroyed");
    }
  }

  function formatTime(ms) {
    const totalSeconds = Math.max(0, Math.ceil(ms / 1000));
    const minutes = Math.floor(totalSeconds / 60);
    const seconds = totalSeconds % 60;

    return `${String(minutes).padStart(2, "0")}:${String(seconds).padStart(2, "0")}`;
  }

  function maskProxy(proxy) {
    if (!proxy) return "—";
    return `${proxy.scheme}://${proxy.host}:${proxy.port}`;
  }

      const ok = await manager.previous();
      setMessage(ok ? "Прокси переключён" : "Переключение не выполнено");
    });

    next.addEventListener("click", async () => {
      setMessage("Переключение...");
      const ok = await manager.next();
      setMessage(ok ? "Прокси переключён" : "Переключение не выполнено");
    });

    now.addEventListener("click", async () => {
      setMessage("Переключение...");
      const ok = await manager.rotate("MANUAL_NEXT");
      setMessage(ok ? "Прокси переключён" : "Переключение не выполнено");
    });

    select.addEventListener("change", async () => {
      const index = Number(select.value);
      setMessage("Активация...");
      const ok = await manager.setProxy(index, "MANUAL_NEXT");
      setMessage(ok ? "Прокси активирован" : "Не удалось активировать прокси");
    });

    test.addEventListener("click", async () => {
      setMessage("Проверка прокси...");
      const result = await manager.testProxy();

      if (result.ok) {
        setMessage(`ACTIVE — IP: ${result.ip || "unknown"}`);
      } else {
        setMessage(`${result.status || "ERROR"} — ${result.error || ""}`);
      }
    });

    add.addEventListener("click", () => {
      const input = prompt(
        "Введите прокси:\nhttp://IP:PORT:LOGIN:PASSWORD\n\nИли:\nhttp://IP:PORT"
      );

      if (!input) return;

      try {
        const proxy = manager.addProxy(input);
        setMessage(`Добавлен: ${maskProxy(proxy)}`);
      } catch (error) {
        manager.error("Add proxy error:", error);
        setMessage(`Ошибка: ${error.message}`);
      }
    });

    remove.addEventListener("click", async () => {
      if (!manager.currentProxy) return;

      const label = maskProxy(manager.currentProxy);
      if (!confirm(`Удалить текущий прокси ${label}?`)) return;

      const deletedIndex = manager.currentProxyIndex;
      const ok = await manager.deleteCurrentProxy();

      if (!ok) {
        setMessage("Не удалось удалить прокси");
        return;
      }

      setMessage(
        manager.currentProxy
          ? `Удалён Proxy ${deletedIndex + 1}`
          : "Список прокси пуст"
      );
    });

    toggle.addEventListener("click", () => {
      if (manager.autoRotation) {
        manager.stop();
        setMessage("Автоматическая ротация выключена");
      } else {
        manager.startAuto();
        setMessage("Автоматическая ротация включена");
      }
    });

    return root;
  }

  async function main() {
    const bridge = new ProxyExtensionBridge();
    const manager = new ProxyRotationManager(PROXY_LIST, bridge);

    createUI(manager);
    await manager.init();

    window.ProxyRotationManager = manager;
    window.ProxyExtensionBridge = bridge;

    window.addEventListener("beforeunload", () => {
      manager.destroy();
    });
  }

  main().catch(error => {
    console.error("[Proxy Rotation] Fatal initialization error:", error);
  });
})();
